% --- Correctif babel-arabic + \part/\chapter ---------------------------------
\makeatletter
\AtBeginDocument{%
  \let\bidifix@chapter\@chapter
  \def\@chapter[#1]#2{%
    \bidifix@chapter[#1]{#2}%
    \expandafter\ifx\@chapapp\appendixname
      \def\cref@chapter@alias{appendix}%
    \fi
    \def\@currentcounter{chapter}%
    \protected@edef\@currentlabel{\csname p@chapter\endcsname\thechapter}%
    \protected@edef\cref@currentlabel{%
      [\expandafter\ifx\@chapapp\appendixname appendix\else chapter\fi]%
      [\the\value{chapter}][]\thechapter}%
  }%
  \let\bidifix@part\@part
  \def\@part[#1]#2{%
    \bidifix@part[#1]{#2}%
    \def\@currentcounter{part}%
    \protected@edef\@currentlabel{\csname p@part\endcsname\thepart}%
    \protected@edef\cref@currentlabel{[part][\the\value{part}][]\thepart}%
  }%
}
\makeatother
